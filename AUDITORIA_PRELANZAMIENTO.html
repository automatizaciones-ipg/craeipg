Bloque <head> Completo para Layout.astro
Cómo usar: Reemplazar el bloque <head> actual de src/layouts/Layout.astro con este. Acepta props: title, description, keywords, ogImage, canonicalURL.
---
// src/layouts/Layout.astro — props extendidos
interface Props {
  title: string;
  description?: string;
  keywords?: string;
  ogImage?: string;
}
const { title, description = "Portal de recursos académicos del CRAE — Instituto Profesional IPG",
        keywords = "recursos académicos, IPG, CRAE, técnicas de estudio, APA, Moodle",
        ogImage = "/og-image.png" } = Astro.props;
const siteUrl = "https://craeipg.automatizaciones-730.workers.dev";
const canonicalURL = new URL(Astro.url.pathname, siteUrl).toString();
const ogImageURL = new URL(ogImage, siteUrl).toString();
---
